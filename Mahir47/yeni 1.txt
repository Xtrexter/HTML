<script>
(() => {
  let userRegistered = false;
  let currentUsername = null;
  let balance = 10000;

  const welcomeScreen = document.getElementById('welcome-screen');
  const modalContainer = document.getElementById('modal-container');
  const accountCloseBtn = document.getElementById('account-close-btn');
  const createAccountForm = document.getElementById('create-account-form');
  const loginForm = document.getElementById('login-form');
  const createAccountMessage = document.getElementById('create-account-message');
  const loginMessage = document.getElementById('login-message');
  const casinoContainer = document.getElementById('casino-container');
  const balanceDisplay = document.getElementById('balance-display');
  const betAmountEl = document.getElementById('bet-amount');

  document.getElementById('create-account-btn').addEventListener('click', () => {
    modalContainer.style.display = 'flex';
    createAccountForm.style.display = 'block';
    loginForm.style.display = 'none';
  });

  document.getElementById('login-btn').addEventListener('click', () => {
    modalContainer.style.display = 'flex';
    createAccountForm.style.display = 'none';
    loginForm.style.display = 'block';
  });

  accountCloseBtn.addEventListener('click', () => {
    modalContainer.style.display = 'none';
  });

  document.getElementById('show-login-form-btn').addEventListener('click', () => {
    createAccountForm.style.display = 'none';
    loginForm.style.display = 'block';
  });

  document.getElementById('show-create-form-btn').addEventListener('click', () => {
    loginForm.style.display = 'none';
    createAccountForm.style.display = 'block';
  });

  createAccountForm.addEventListener('submit', (e) => {
    e.preventDefault();
    const username = document.getElementById('create-username').value.trim();
    const email = document.getElementById('create-email').value.trim();
    const password = document.getElementById('create-password').value;

    let users = JSON.parse(localStorage.getItem('mahox_users')) || {};
    if (users[username]) {
      createAccountMessage.textContent = 'Kullanıcı adı zaten mevcut.';
      createAccountMessage.className = 'error';
      return;
    }

    users[username] = { username, email, password, balance: 10000 };
    localStorage.setItem('mahox_users', JSON.stringify(users));

    currentUsername = username;
    userRegistered = true;
    balance = users[username].balance;

    updateUI();
    modalContainer.style.display = 'none';
  });

  loginForm.addEventListener('submit', (e) => {
    e.preventDefault();
    const username = document.getElementById('login-username').value.trim();
    const password = document.getElementById('login-password').value;

    let users = JSON.parse(localStorage.getItem('mahox_users')) || {};
    const user = users[username];

    if (!user || user.password !== password) {
      loginMessage.textContent = 'Geçersiz kullanıcı adı veya şifre.';
      loginMessage.className = 'error';
      return;
    }

    currentUsername = username;
    userRegistered = true;
    balance = user.balance;

    updateUI();
    modalContainer.style.display = 'none';
  });

  function updateUI() {
    balanceDisplay.textContent = `Balance: ${balance} 🪙`;
    welcomeScreen.style.display = 'none';
    casinoContainer.style.display = 'block';
  }
})();
</script>
